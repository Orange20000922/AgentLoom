cmake_minimum_required(VERSION 3.20)
project(WholeArchiveRegression LANGUAGES CXX)

if(MSVC)
    add_compile_options(/utf-8)
endif()

# registration.cpp 无显式符号引用，必须通过整库链接保留。
if(WHOLE_ARCHIVE_PRODUCER)
    add_library(config STATIC registry.cpp registration.cpp)
    install(TARGETS config EXPORT ConfigTargets ARCHIVE DESTINATION lib)
    install(EXPORT ConfigTargets NAMESPACE AgentLoom:: DESTINATION lib/cmake/fixture)
    install(FILES "${AGENTLOOM_SOURCE_DIR}/cmake/AgentLoomWholeArchive.cmake"
        DESTINATION lib/cmake/fixture)
    return()
endif()

include("${FIXTURE_PREFIX}/lib/cmake/fixture/ConfigTargets.cmake")
include("${FIXTURE_PREFIX}/lib/cmake/fixture/AgentLoomWholeArchive.cmake")
set(BERT_VCPKG_TRIPLET "" CACHE STRING "No external dependency in this fixture")
include("${AGENTLOOM_SOURCE_DIR}/cmake/AgentLoomOptions.cmake")
add_library(local_config STATIC registry.cpp registration.cpp)
add_library(Local::config ALIAS local_config)
add_library(Imported::config ALIAS AgentLoom::config)

enable_testing()
foreach(mode IN ITEMS local imported internal forward)
    if(mode STREQUAL "imported")
        set(config Imported::config)
    elseif(mode STREQUAL "forward")
        set(config Forward::config)
    else()
        set(config Local::config)
    endif()
    add_library(${mode}_indirect INTERFACE)
    target_link_libraries(${mode}_indirect INTERFACE "${config}")
    add_executable(${mode}_consumer main.cpp)
    target_link_libraries(${mode}_consumer PRIVATE "${config}" ${mode}_indirect)
    if(mode STREQUAL "internal" OR mode STREQUAL "forward")
        link_whole_archive(${mode}_consumer "${config}")
    else()
        agentloom_link_whole_archive(${mode}_consumer "${config}")
    endif()
    # 同一真实库从 ALIAS/真实名称重复调用也必须幂等。
    if(mode STREQUAL "imported")
        agentloom_link_whole_archive(${mode}_consumer AgentLoom::config)
    elseif(NOT mode STREQUAL "forward")
        agentloom_link_whole_archive(${mode}_consumer local_config)
    endif()
    add_test(NAME ${mode}_consumer COMMAND ${mode}_consumer)
    set_tests_properties(${mode}_consumer PROPERTIES PASS_REGULAR_EXPRESSION "registered=1")
endforeach()

# 对齐 Foundation 先引用、ModelTargets 后定义配置库的实际源码构建顺序。
add_library(forward_config STATIC registry.cpp registration.cpp)
add_library(Forward::config ALIAS forward_config)
